const EASE = {
  // smoothstep with calculus derivative for velocity
  smooth: t => t*t*(3-2*t),
  smoother: t => t*t*t*(t*(t*6-15)+10),
  spring: (x,target,v,k=0.08,d=0.82) => {
    const a = (target - x)*k - v*d;
    return {v: v + a, x: x + v + a};
  }
};

/* ---------- Global Ambient: dust motes + glyph grid + cursor trail ---------- */
(function(){
  const canvas = document.getElementById('global-ambient');
  const ctx = canvas.getContext('2d');
  let W,H, dpr;
  const motes = [];
  const glyphs = [];
  const trail = [];
  let lastMouse = {x:0,y:0,time:0};
  let cursorSpeed = 0;
  const glyphPool = ['က','ည','ီ','အတၢ်','ကရၢ','ကရိ','လၢ','ကၠီ','ၤ','မဲ','ၡ�','ရကၢ','K','O','A','၀','၁','၂','၃','၄','၅','၆','●','○','·'];
  
  function resize(){
    dpr = window.devicePixelRatio||1;
    W = window.innerWidth; H = window.innerHeight;
    canvas.width = W*dpr; canvas.height = H*dpr;
    canvas.style.width = W+'px'; canvas.style.height = H+'px';
    ctx.setTransform(dpr,0,0,dpr,0,0);
  }
  function init(){
    resize();
    for(let i=0;i<120;i++){
      motes.push({x:Math.random()*W, y:Math.random()*H, vx:(Math.random()-0.5)*0.3, vy:(Math.random()-0.5)*0.3, r:Math.random()*1.6+0.3, alpha:Math.random()*0.5+0.2, phase:Math.random()*Math.PI*2});
    }
    for(let i=0;i<90;i++){
      glyphs.push({x:Math.random()*W, y:Math.random()*H, char:glyphPool[Math.floor(Math.random()*glyphPool.length)], alpha:Math.random()*0.12+0.04, size: 9+Math.random()*10, phase:Math.random()*6, lastChange:0});
    }
  }
  window.addEventListener('mousemove', e=>{
    const now = performance.now();
    const dx = e.clientX - lastMouse.x;
    const dy = e.clientY - lastMouse.y;
    const dt = Math.max(1, now-lastMouse.time);
    const speed = Math.hypot(dx,dy)/dt*16;
    cursorSpeed = cursorSpeed*0.85 + speed*0.15;
    lastMouse = {x:e.clientX, y:e.clientY, time:now};
    trail.push({x:e.clientX, y:e.clientY, life:1, time:now});
    if(trail.length>24) trail.shift();
    const dot = document.getElementById('cursor-dot');
    dot.style.left = e.clientX-4+'px'; dot.style.top = e.clientY-4+'px'; dot.style.opacity=0.9;
    clearTimeout(dot._t); dot._t=setTimeout(()=>dot.style.opacity=0, 600);
  });
  window.addEventListener('resize', resize);

  let t=0;
  function frame(){
    t+=0.016;
    ctx.clearRect(0,0,W,H);
    // subtle gradient wash
    const grad = ctx.createRadialGradient(W*0.5, H*0.2, 0, W*0.5, H*0.2, W*1.2);
    grad.addColorStop(0,'rgba(35,59,120,0.08)'); grad.addColorStop(1,'rgba(0,0,0,0)');
    ctx.fillStyle=grad; ctx.fillRect(0,0,W,H);

    // glyph grid with randomization frequency tied to cursor speed
    const freqBase = 0.006;
    const freq = freqBase + cursorSpeed*0.012;
    glyphs.forEach(g=>{
      if(Math.random()<freq){
        g.char = glyphPool[Math.floor(Math.random()*glyphPool.length)];
      }
      // ambient drift side to side ever so slightly
      const driftX = Math.sin(t*0.3 + g.phase)*2.2;
      const driftY = Math.cos(t*0.22 + g.phase)*1.1;
      ctx.globalAlpha = g.alpha*(0.8+Math.sin(t+g.phase)*0.2);
      ctx.font = g.size+'px "Instrument Serif"';
      ctx.fillStyle = '#eadbb0';
      ctx.fillText(g.char, g.x+driftX, g.y+driftY);
    });
    ctx.globalAlpha=1;

    // motes buttery physics
    motes.forEach(m=>{
      m.vx += (Math.sin(t*0.2+m.phase)*0.002);
      m.vy += (Math.cos(t*0.15+m.phase)*0.002);
      m.x += m.vx; m.y += m.vy;
      if(m.x<0) m.x=W; if(m.x>W) m.x=0; if(m.y<0) m.y=H; if(m.y>H) m.y=0;
      ctx.beginPath(); ctx.globalAlpha=m.alpha; ctx.fillStyle='#eadbb0';
      ctx.arc(m.x+Math.sin(t*0.5+m.phase)*1.5, m.y, m.r,0,Math.PI*2); ctx.fill();
    });
    ctx.globalAlpha=1;

    // cursor trail navigation
    trail.forEach((p,i)=>{
      const age = (performance.now()-p.time)/900;
      p.life = 1-age;
      if(p.life<=0) return;
      const r = (1-p.life)*22 + 3;
      ctx.beginPath();
      ctx.globalAlpha = p.life*0.18;
      const g = ctx.createRadialGradient(p.x,p.y,0,p.x,p.y,r);
      g.addColorStop(0,'rgba(234,219,176,0.9)'); g.addColorStop(1,'rgba(234,219,176,0)');
      ctx.fillStyle=g; ctx.arc(p.x,p.y,r,0,Math.PI*2); ctx.fill();
    });
    // prune
    for(let i=trail.length-1;i>=0;i--) if(trail[i].life<=0) trail.splice(i,1);

    requestAnimationFrame(frame);
  }
  init(); frame();
})();

function getOutlinePoints(letter, width=300, height=380, step=7){
  const c = document.createElement('canvas'); c.width=width; c.height=height;
  const ctx = c.getContext('2d');
  ctx.fillStyle='black'; ctx.fillRect(0,0,width,height);
  ctx.fillStyle='white'; ctx.font='900 300px "Instrument Serif", serif'; ctx.textAlign='center'; ctx.textBaseline='middle';
  ctx.fillText(letter, width/2, height/2+20);
  const data = ctx.getImageData(0,0,width,height).data;
  const pts=[];
  for(let y=0;y<height;y+=step){
    for(let x=0;x<width;x+=step){
      const idx=(y*width+x)*4;
      if(data[idx]>150){
        // edge check
        let isEdge=false;
        for(let dy=-1;dy<=1;dy++) for(let dx=-1;dx<=1;dx++){
          const nx=x+dx*2, ny=y+dy*2;
          if(nx<0||ny<0||nx>=width||ny>=height){isEdge=true; break;}
          const nidx=(ny*width+nx)*4;
          if(data[nidx]<80){isEdge=true; break;}
        }
        if(isEdge) pts.push({x,y});
      }
    }
  }
  return pts;
}

function buildLetter(box, letter){
  const pts = getOutlinePoints(letter, box.clientWidth||300, box.clientHeight||380, 8);
  box.innerHTML='';
  const elements=[];
  pts.forEach((p,i)=>{
    const span=document.createElement('span');
    span.className='letter-glyph '+(Math.random()>0.5?'karen':'latin');
    span.textContent = glyphPoolKA[Math.floor(Math.random()*glyphPoolKA.length)];
    span.style.left = p.x+'px'; span.style.top = p.y+'px';
    span.dataset.baseX = p.x; span.dataset.baseY = p.y;
    span.dataset.phase = (Math.random()*Math.PI*2).toString();
    span.dataset.freq = (0.3+Math.random()*0.6).toString();
    span.dataset.amp = (0.6+Math.random()*1.8).toString();
    // physics state for buttery calculus
    span._x = p.x; span._y = p.y; span._vx=0; span._vy=0;
    box.appendChild(span);
    elements.push(span);
  });
  return elements;
}

function animateLetters(){
  tGlobal+=0.016;
  const all = [...kGlyphs, ...aGlyphs];
  all.forEach(el=>{

    const phase=parseFloat(el.dataset.phase), freq=parseFloat(el.dataset.freq), amp=parseFloat(el.dataset.amp);
    // ambient ever so slightly side to side - calculus driven target
    const targetX = baseX + Math.sin(tGlobal*freq + phase)*amp;
    const targetY = baseY + Math.cos(tGlobal*freq*0.6 + phase)*amp*0.35;
    // spring to target for buttery
    const k=0.12, d=0.78;
    const ax = (targetX - el._x)*k - el._vx*d;
    const ay = (targetY - el._y)*k - el._vy*d;
    el._vx+=ax; el._vy+=ay;
    el._x+=el._vx; el._y+=el._vy;
    el.style.transform=`translate3d(${el._x-baseX}px,${el._y-baseY}px,0)`;
  });
  requestAnimationFrame(animateLetters);
}

class MorphSet{
  constructor(container, count=18){
    this.container=container; this.glyphs=[];
    for(let i=0;i<count;i++){
      const s=document.createElement('span'); s.className='morph-glyph'; s.textContent=glyphPoolKA[Math.floor(Math.random()*glyphPoolKA.length)];
      s._x=Math.random()*10-5; s._y=Math.random()*10-5; s._vx=0; s._vy=0; s._tx=0; s._ty=0;
      container.appendChild(s); this.glyphs.push(s);
    }
  }
  morphTo(points){
    // assign each glyph to nearest point with shuffle
    const shuffled=[...points].sort(()=>Math.random()-0.5);
    this.glyphs.forEach((g,i)=>{
      const p=shuffled[i%shuffled.length]||{x:0,y:0};
      g._tx=p.x; g._ty=p.y;
    });
  }
  update(){
    this.glyphs.forEach(g=>{
      const ax=(g._tx-g._x)*0.14 - g._vx*0.82;
      const ay=(g._ty-g._y)*0.14 - g._vy*0.82;
      g._vx+=ax; g._vy+=ay; g._x+=g._vx; g._y+=g._vy;
      g.style.transform=`translate3d(${g._x}px,${g._y}px,0)`;
    });
  }
}
