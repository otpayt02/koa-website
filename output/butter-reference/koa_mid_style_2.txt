function getNumeralPoints(numeral, size=26){

  ctx.fillStyle='black'; ctx.fillRect(0,0,size,size);
  ctx.fillStyle='white'; ctx.font='700 '+(size*0.8)+'px "Instrument Serif"'; ctx.textAlign='center'; ctx.textBaseline='middle';
  ctx.fillText(numeral, size/2, size/2+1);
  const data=ctx.getImageData(0,0,size,size).data; const pts=[];
  for(let y=0;y<size;y+=3) for(let x=0;x<size;x+=3){ if(data[(y*size+x)*4]>120) pts.push({x:x-size/2, y:y-size/2}); }
  return pts;
}
class MorphSet{
  constructor(container,count=16){ this.container=container; this.glyphs=[]; for(let i=0;i<count;i++){ const s=document.createElement('span'); s.className='morph-glyph'; s.textContent=glyphPoolKA[Math.floor(Math.random()*glyphPoolKA.length)]; s._x=Math.random()*10-5; s._y=Math.random()*10-5; s._vx=0; s._vy=0; s._tx=0; s._ty=0; container.appendChild(s); this.glyphs.push(s); } }
  morphTo(points){ const shuffled=[...points].sort(()=>Math.random()-0.5); this.glyphs.forEach((g,i)=>{ const p=shuffled[i%shuffled.length]||{x:0,y:0}; g._tx=p.x; g._ty=p.y; }); }
  update(){ this.glyphs.forEach(g=>{ const ax=(g._tx-g._x)*0.14 - g._vx*0.82; const ay=(g._ty-g._y)*0.14 - g._vy*0.82; g._vx+=ax; g._vy+=ay; g._x+=g._vx; g._y+=g._vy; g.style.transform=`translate3d(${g._x}px,${g._y}px,0)`; }); }
}
const morphSets=[];

(function morphLoop(){ morphSets.forEach(ms=>ms.update()); requestAnimationFrame(morphLoop); })();

sections.forEach(id=>{ const el=document.getElementById(id); if(el) spyObserver.observe(el); });
